% Available evidence and distinct inferential targets in LD-PdM.
% Used in Section 2; standalone previews use sources/tikz/diagram.tex.
\begin{tikzpicture}[
  font=\small,
  evidence/.style={draw, rounded corners=2pt, align=center,
    text width=3.85cm, minimum height=1.05cm,
    inner xsep=5pt, inner ysep=4pt},
  heading/.style={font=\small\bfseries, align=center, text width=3.85cm},
  inference/.style={evidence, text width=2.75cm, minimum height=2.7cm},
  note/.style={font=\footnotesize, align=center},
  inputlink/.style={-Latex, thick, dashed},
  inferencelink/.style={-Latex, thick, dotted}
]
\node[heading] (available) {Available evidence};
\node[heading, right=5.8cm of available] (targets) {Quantities to be inferred};

\node[evidence, below=5mm of available] (proxy) {\textbf{Usage proxies}\\traffic, schedules,\\ticket flow};
\node[evidence, below=4mm of proxy] (context) {\textbf{Context}\\location, time,\\operating regime};
\node[evidence, below=4mm of context] (history) {\textbf{Service history}\\interventions and inspection records};
\node[evidence, below=4mm of history] (events) {\textbf{Failure-event records}\\possibly sparse or incomplete};

\node[evidence] (usage) at (targets |- proxy) {\textbf{Usage and load}\\device-specific activity\\and burden};
\node[evidence] (exposure) at (targets |- context) {\textbf{Exposure}\\accumulated\\operational burden};
\node[evidence] (condition) at (targets |- history) {\textbf{Condition or degradation}\\not determined by exposure alone};
\node[evidence] (probability) at (targets |- events) {\textbf{Event probability}\\specified event and horizon};

\coordinate (middle) at ($(context)!0.5!(history)$);
\node[inference] (beliefs) at ($(middle)!0.5!(targets |- middle)$) {
  \textbf{Probabilistic inference}\\[2mm]
  Engineering assumptions\\[1mm]
  Belief updating\\[1mm]
  Uncertainty propagation
};
\node[note, text width=2.8cm, above=5mm of beliefs] {Weak or unavailable\\condition-sensitive\\channels};

\foreach \source/\offset in {proxy/10,context/3,history/-3,events/-10} {
  \draw[inputlink] (\source.east) to[out=0,in=180] ([yshift=\offset mm]beliefs.west);
}
\foreach \target/\offset in {usage/10,exposure/3,condition/-3,probability/-10} {
  \draw[inferencelink] ([yshift=\offset mm]beliefs.east) to[out=0,in=180] (\target.west);
}

\node[note, text width=13.1cm, below=6mm of events.south west, anchor=north west] {
  Maintenance choices also depend on consequences, feasible actions and service constraints.\\
  Proxy evidence alone does not establish physical condition or calibrated event probabilities.
};
\end{tikzpicture}
